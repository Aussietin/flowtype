# Windows version resource for flowtype.exe (PyInstaller EXE(version=...)).
# Shows up under the exe's Properties -> Details and in Task Manager. An exe
# with no company/product/description metadata is one of the cheapest signals
# AV heuristics and buyers use to call a binary "unknown".

# in installer.iss -- test_version.py fails the build if they drift.

    filevers=(1, 0, 0, 0),
    prodvers=(1, 0, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '040904B0',

         StringStruct('OriginalFilename', 'flowtype.exe'),
         StringStruct('ProductName', 'flowtype'),
         StringStruct('ProductVersion', '1.0.0.0')])
    ]),
    VarFileInfo([VarStruct('Translation', [0x0409, 1200])])
  ]
)
